cmake_minimum_required(VERSION 3.25)
project(FCodeWindowsComputerUse LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows Computer Use requires MSVC and the Windows SDK")
endif()
string(LENGTH "${FCODE_HOST_IMAGE_SHA256}" FCODE_HOST_IMAGE_SHA256_LENGTH)
if(NOT FCODE_HOST_IMAGE_SHA256 MATCHES "^[0-9a-f]+$" OR NOT FCODE_HOST_IMAGE_SHA256_LENGTH EQUAL 64)
  message(FATAL_ERROR "A verified host executable SHA-256 is required")
endif()
if(NOT FCODE_HOST_ROLE STREQUAL "prototype-image" AND NOT FCODE_HOST_ROLE STREQUAL "electron-main")
  message(FATAL_ERROR "An explicit native host role policy is required")
endif()
if(FCODE_LEASE_TEST_SCOPE AND NOT FCODE_HOST_ROLE STREQUAL "prototype-image")
  message(FATAL_ERROR "Lease test scopes are forbidden for GUI Main hosts")
endif()
configure_file(HostPolicy.h.in HostPolicy.h @ONLY)
add_executable(fcode-cua-helper Main.cpp NativeCrypto.cpp HostIdentity.cpp HostRole.cpp DesktopLease.cpp Supervisor.cpp WindowRegistry.cpp WindowLifecycle.cpp TargetAccess.cpp WindowCapture.cpp WindowPng.cpp ObservationStore.cpp UiaObserver.cpp UiaObserverClient.cpp WindowMessageInput.cpp DesktopCapture.cpp DesktopState.cpp DesktopInput.cpp)
target_compile_features(fcode-cua-helper PRIVATE cxx_std_20)
target_compile_definitions(fcode-cua-helper PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(fcode-cua-helper PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(fcode-cua-helper PRIVATE /W4 /permissive- /EHsc /guard:cf /utf-8 /MP2)
target_link_options(fcode-cua-helper PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
set_property(TARGET fcode-cua-helper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(fcode-cua-helper PRIVATE windowsapp user32 advapi32 bcrypt dwmapi d3d11 dxgi windowscodecs crypt32 ole32 oleaut32 uiautomationcore wbemuuid shell32)
